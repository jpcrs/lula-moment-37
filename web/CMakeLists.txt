cmake_minimum_required(VERSION 3.24)
project(moment37 C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(GAME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../3sx")
file(GLOB_RECURSE GAME_SOURCES "${GAME_ROOT}/src/*.c")
list(FILTER GAME_SOURCES EXCLUDE REGEX "/(platform|test|imgui)/")
list(FILTER GAME_SOURCES EXCLUDE REGEX "/(main|game|afs|args|resources|input|rom|sha256|keymap|config|config_helpers)\\.c$")
list(FILTER GAME_SOURCES EXCLUDE REGEX "/(menu|pause|plpat|tate00|sound3rd)\\.c$")
list(FILTER GAME_SOURCES EXCLUDE REGEX "/hitcheck\\.c$")
list(FILTER GAME_SOURCES EXCLUDE REGEX "/aboutspr\\.c$")
list(FILTER GAME_SOURCES EXCLUDE REGEX "/sc_sub\\.c$")
add_executable(engine ${GAME_SOURCES} "${GAME_ROOT}/src/port/io/afs.c" native/web_app.c native/moment_runtime.c build/generated/main_scene.c build/generated/game_scene.c build/generated/plpat_scene.c build/generated/hitcheck_scene.c build/generated/stage_scene.c build/generated/sound_scene.c build/generated/aboutspr_scene.c build/generated/sc_sub_scene.c native/web_support.c native/webgl_renderer.c "${GAME_ROOT}/src/platform/video/sdl_generic/sdl_generic_renderer.c" "${GAME_ROOT}/src/platform/netplay/game_state.c")
target_include_directories(engine PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native" "${GAME_ROOT}/src" "${GAME_ROOT}/src/sdk" "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/SDL/include" "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/SDL/build-web/include-config-release")
target_compile_definitions(engine PRIVATE RELEASE=1 SOUND_ENABLED=1 CRS_VIDEO_DRIVER_SDL_GENERIC=1 CRS_VIDEO_DRIVER_OPENGL=1)
target_compile_options(engine PRIVATE -O2 --use-port=zlib -fno-strict-aliasing -Wno-incompatible-pointer-types -Wno-int-conversion -Wno-pointer-to-int-cast -Wno-deprecated-non-prototype)
target_link_libraries(engine PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/SDL/build-web/libSDL3.a" m)
target_link_options(engine PRIVATE -O2 -sEMULATE_FUNCTION_POINTER_CASTS=1 --emit-symbol-map --use-port=zlib -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=134217728 -sMAXIMUM_MEMORY=1073741824 -sSTACK_SIZE=4194304 -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sMODULARIZE=1 -sEXPORT_NAME=createMoment37 -sENVIRONMENT=web,node -sFORCE_FILESYSTEM=1 "-sEXPORTED_RUNTIME_METHODS=['FS','HEAPU8']" "-sEXPORTED_FUNCTIONS=['_web_init','_web_step','_web_input','_web_start','_web_status','_web_value','_web_animation','_web_render','_web_reset','_web_original']" "SHELL:--preload-file ${CMAKE_CURRENT_SOURCE_DIR}/assets/challenge.afs@/SF33RD.AFS" "SHELL:--preload-file ${CMAKE_CURRENT_SOURCE_DIR}/native/shaders@/shaders/opengl" "SHELL:--preload-file ${CMAKE_CURRENT_SOURCE_DIR}/dist/characters@/characters" "SHELL:--preload-file ${GAME_ROOT}/assets@/assets" --no-entry)
set_target_properties(engine PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist")

file(GLOB SHADER_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/native/shaders/*.glsl")
set_property(TARGET engine APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/challenge.afs" ${SHADER_INPUTS})
file(GLOB CHARACTER_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/dist/characters/*")
set_property(TARGET engine APPEND PROPERTY LINK_DEPENDS ${CHARACTER_INPUTS})

add_custom_command(TARGET engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E rename "${CMAKE_CURRENT_SOURCE_DIR}/dist/engine.js.symbols" "${CMAKE_CURRENT_BINARY_DIR}/engine.symbols")
